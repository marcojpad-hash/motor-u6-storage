import math from typing import Any, Dict, List, Sequence, Tuple

def _rank(board: Dict[str, float]) -> Dict[str, int]: orden =
sorted(board, key=lambda k: -board[k]) return {k: i + 1 for i, k in
enumerate(orden)}

def _corr(a: Sequence[float], b: Sequence[float]): n = len(a) if n < 2:
return None ma = sum(a) / n mb = sum(b) / n va = sum((x - ma) ** 2 for x
in a) vb = sum((y - mb) ** 2 for y in b) if va <= 1e-18 or vb <= 1e-18:
return 0.0 cov = sum((x - ma) * (y - mb) for x, y in zip(a, b)) return
cov / math.sqrt(va * vb)

def analizar( cortes: Sequence[Tuple[str, Dict[str, float]]],
nota_umbral: float | None = None, ) -> Dict[str, Any]: ids = [c[0] for c
in cortes] boards = [c[1] for c in cortes] nombres =
sorted(set().union(*[b.keys() for b in boards]))

    serie: Dict[str, List[float]] = {n: [] for n in nombres}
    rangos: Dict[str, List[int]] = {n: [] for n in nombres}
    for b in boards:
        rk = _rank(b)
        for n in nombres:
            serie[n].append(round(float(b.get(n, 0.5)), 6))
            rangos[n].append(rk.get(n, len(nombres) + 1))

    # un vector por tramo: TODAS las Δ
    vectores = []
    deltas_por_h: Dict[str, List[float]] = {n: [] for n in nombres}
    for i in range(1, len(boards)):
        a, b = boards[i - 1], boards[i]
        mov = []
        for n in nombres:
            d = float(b.get(n, 0.5)) - float(a.get(n, 0.5))
            deltas_por_h[n].append(d)
            mov.append({
                "h": n,
                "d": round(d, 6),
                "P_de": round(float(a.get(n, 0.5)), 6),
                "P_a": round(float(b.get(n, 0.5)), 6),

                "rango_a": rangos[n][i],
                "drango": rangos[n][i - 1] - rangos[n][i],
            })

        vectores.append({
            "vector": f"{ids[i-1]}→{ids[i]}",
